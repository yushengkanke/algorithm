# 板子  
```c++
struct DSU {
    vector<int> fa, p, e, f;
    DSU(int n) {
        fa.resize(n + 1);//父节点
        iota(fa.begin(), fa.end(), 0);//按顺序和间隔为一给数组赋值，每赋值一个数，value(代码中的0)就会加一
        p.resize(n + 1, 1);
        e.resize(n + 1);
        f.resize(n + 1);
    }
    int get(int x) {
        while (x != fa[x]) {
            x = fa[x] = fa[fa[x]];
        }
        return x;
    }
    bool merge(int x, int y) { // 设x是y的祖先
        if (x == y) f[get(x)] = 1;
        x = get(x), y = get(y);
        e[x]++;
        if (x == y) return false;
        if (x < y) swap(x, y); // 将编号小的合并到大的上
        fa[y] = x;
        f[x] |= f[y], p[x] += p[y], e[x] += e[y];
        return true;
    }
    bool same(int x, int y) {
        return get(x) == get(y);
    }
    bool F(int x) { // 判断连通块内是否存在自环
        return f[get(x)];
    }
    int size(int x) { // 输出连通块中点的数量
        return p[get(x)];
    }
    int E(int x) { // 输出连通块中边的数量
        return e[get(x)];
    }
};
```
注意，在查询父节点的时候会进行路径压缩，和启发式合并，原图会被破坏，仅用作分成集合  
当使用启发式合并时，n个元素和m个查询时，时间复杂度为O(mlog(n))   
# 查找无向图中是否有环 
核心原理：
遍历所有边 (u, v)。如果 u 和 v 已经在同一个连通分量中（即它们的根节点相同），那么加上这条边一定会形成环；如果不在同一个分量，则合并 u 和 v。  
```c++
struct DSU {
    vector<int> parent, rank;
    DSU(int n) {
        parent.resize(n + 1);
        rank.resize(n + 1, 0);
        for (int i = 1; i <= n; i++) parent[i] = i;
    }
    int find(int x) {
        return parent[x] == x ? x : parent[x] = find(parent[x]);
    }
    bool unite(int x, int y) {
        int rx = find(x), ry = find(y);
        if (rx == ry) return false; // 已在同一集合，加上这条边会成环
        if (rank[rx] < rank[ry]) swap(rx, ry);
        parent[ry] = rx;
        if (rank[rx] == rank[ry]) rank[rx]++;
        return true;
    }
};

bool hasCycle(int n, vector<pair<int,int>>& edges) {
    DSU dsu(n);
    for (auto &[u, v] : edges) {
        if (!dsu.unite(u, v)) return true; // 合并失败 => 存在环
    }
    return false;
}
```
