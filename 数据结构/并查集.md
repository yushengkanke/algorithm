# 板子  
```c++
struct DSU {
    vector<int> fa, p, e, f;
    DSU(int n) {
        fa.resize(n + 1);//父节点
        iota(fa.begin(), fa.end(), 0);//按顺序和间隔为一给数组赋值，每赋值一个数，value(代码中的0)就会加一
        p.resize(n + 1, 1);
        e.resize(n + 1);
        f.resize(n + 1);
    }
    int get(int x) {
        while (x != fa[x]) {
            x = fa[x] = fa[fa[x]];
        }
        return x;
    }
    bool merge(int x, int y) { // 设x是y的祖先
        if (x == y) f[get(x)] = 1;
        x = get(x), y = get(y);
        e[x]++;
        if (x == y) return false;
        if (x < y) swap(x, y); // 将编号小的合并到大的上
        fa[y] = x;
        f[x] |= f[y], p[x] += p[y], e[x] += e[y];
        return true;
    }
    bool same(int x, int y) {
        return get(x) == get(y);
    }
    bool F(int x) { // 判断连通块内是否存在自环
        return f[get(x)];
    }
    int size(int x) { // 输出连通块中点的数量
        return p[get(x)];
    }
    int E(int x) { // 输出连通块中边的数量
        return e[get(x)];
    }
};
```
注意，在查询父节点的时候会进行路径压缩，和启发式合并，原图会被破坏，仅用作分成集合  
当使用启发式合并时，n个元素和m个查询时，时间复杂度为O(mlog(n))
