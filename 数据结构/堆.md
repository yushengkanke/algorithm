# 定义 
1.完全二叉树
2.子节点永远比父节点小或相等（小根堆）或者大或相等（大根堆）
# 创建堆 
开一个vector<>，其中对于父节点n来说，2n和2n+1为它的子节点
# 插入数据  
先把数据读入末尾，然后依次与n/2比较，如果大或者小则与其交换位置，直到符合大根堆或小根堆的要求
```C++
const int N = 1e5 + 10;   // 堆容量
int h[N], sz;

// 大根堆用 >，小根堆用 <，改这一行即可
bool cmp(int a, int b) { return a > b; }   // 大根堆

void up(int i) {
    while (i > 1 && cmp(h[i], h[i >> 1])) {
        swap(h[i], h[i >> 1]);
        i >>= 1;
    }
}

void down(int i) {
    while ((i << 1) <= sz) {
        int j = i << 1;
        if (j < sz && cmp(h[j + 1], h[j])) ++j;   // 选更优的儿子
        if (!cmp(h[j], h[i])) break;
        swap(h[i], h[j]);
        i = j;
    }
}

void push(int x) { h[++sz] = x; up(sz); }
void pop()       { h[1] = h[sz--]; down(1); }
int  top()       { return h[1]; }
bool empty()     { return sz == 0; }
int  size()      { return sz; }
//可以通过单独开一个数组维护每一个特定某个物体的数值在堆中的位置，堆中应该有相应的改动
```
