# 非递归方式遍历二叉树
时间复杂度同样是O(n)  
看起来就非常的没用，但是考试会考。  
使用栈来实现先序中序和后序。  
```C
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

#define MAXN 1000   /* 栈/队列的最大容量，实际应用可改为动态扩容 */

/* ---------- 二叉树节点 ---------- */
typedef struct TreeNode {
    int val;
    struct TreeNode *left;
    struct TreeNode *right;
} TreeNode;

TreeNode *newNode(int val) {
    TreeNode *n = (TreeNode *)malloc(sizeof(TreeNode));
    n->val = val;
    n->left = n->right = NULL;
    return n;
}

/* ---------- 顺序栈（存节点指针） ---------- */
typedef struct {
    TreeNode *data[MAXN];
    int top;                /* top == -1 表示空 */
} Stack;

void initStack(Stack *s)          { s->top = -1; }
bool stackEmpty(Stack *s)         { return s->top == -1; }
void push(Stack *s, TreeNode *p)  { s->data[++s->top] = p; }
TreeNode *pop(Stack *s)           { return s->data[s->top--]; }
TreeNode *peek(Stack *s)          { return s->data[s->top]; }

/* ---------- 循环队列（存节点指针） ---------- */
typedef struct {
    TreeNode *data[MAXN];
    int front, rear;
} Queue;

void initQueue(Queue *q)          { q->front = q->rear = 0; }
bool queueEmpty(Queue *q)         { return q->front == q->rear; }
bool queueFull(Queue *q)          { return (q->rear + 1) % MAXN == q->front; }
void enqueue(Queue *q, TreeNode *p) {
    if (queueFull(q)) return;
    q->data[q->rear] = p;
    q->rear = (q->rear + 1) % MAXN;
}
TreeNode *dequeue(Queue *q)       { return q->data[q->front++]; }

/* ---------- 1. 前序遍历：根 -> 左 -> 右 ---------- */
void preorder(TreeNode *root) {
    if (!root) return;
    Stack s;
    initStack(&s);
    push(&s, root);
    while (!stackEmpty(&s)) {
        TreeNode *node = pop(&s);
        printf("%d ", node->val);
        if (node->right) push(&s, node->right);  /* 先压右 */
        if (node->left)  push(&s, node->left);   /* 后压左，左先出 */
    }
}

/* ---------- 2. 中序遍历：左 -> 根 -> 右 ---------- */
void inorder(TreeNode *root) {
    Stack s;
    initStack(&s);
    TreeNode *cur = root;
    while (cur || !stackEmpty(&s)) {
        while (cur) {                 /* 一路向左压栈 */
            push(&s, cur);
            cur = cur->left;
        }
        cur = pop(&s);                /* 弹出并访问 */
        printf("%d ", cur->val);
        cur = cur->right;             /* 转向右子树 */
    }
}

/* ---------- 3. 后序遍历：左 -> 右 -> 根（单栈 + lastVisited） ---------- */
void postorder(TreeNode *root) {
    Stack s;
    initStack(&s);
    TreeNode *cur = root, *last = NULL;
    while (cur || !stackEmpty(&s)) {
        while (cur) {
            push(&s, cur);
            cur = cur->left;
        }
        TreeNode *top = peek(&s);
        /* 右孩子存在且尚未访问，则转向右子树 */
        if (top->right && top->right != last) {
            cur = top->right;
        } else {
            printf("%d ", top->val);
            last = pop(&s);           /* 记录最后访问的节点 */
        }
    }
}

/* ---------- 4. 层序遍历：按层从左到右（BFS） ---------- */
void levelorder(TreeNode *root) {
    if (!root) return;
    Queue q;
    initQueue(&q);
    enqueue(&q, root);
    while (!queueEmpty(&q)) {
        TreeNode *node = dequeue(&q);
        printf("%d ", node->val);
        if (node->left)  enqueue(&q, node->left);
        if (node->right) enqueue(&q, node->right);
    }
}

/* ---------- 测试 ---------- */
int main(void) {
    /*
            1
           / \
          2   3
         / \   \
        4   5   6
    */
    TreeNode *root = newNode(1);
    root->left = newNode(2);
    root->right = newNode(3);
    root->left->left = newNode(4);
    root->left->right = newNode(5);
    root->right->right = newNode(6);

    printf("前序: "); preorder(root);   printf("\n");
    printf("中序: "); inorder(root);    printf("\n");
    printf("后序: "); postorder(root);  printf("\n");
    printf("层序: "); levelorder(root); printf("\n");

    return 0;
}
```

# 前序
```C
void preorder(binarytree tree)
{
    if(tree==NULL||tree->size<1)
    {
        return ;
    }
    if(tree->data[1]=='-') //-号为自定义的结束符号
    {
        return ;
    }
    int n = tree->size;
    int *st = (int *)malloc(sizeof(int)*(n+1));
    int top = 0;
    st[top++]=1;
    while(top>0){
        int p = st[--top];
        if(p>n||tree->data[p]=='-')
        {
            continue;
        }
        putchar(tree->data[p]);//进行相关处理
        int l = 2*p;
        int r = 2*p+1;
        //先压右再压左
        if(r<=n&&tree->data[r]!='-')
        {
            st[top++]=r;
        }
        if(l<=n&&tree->data[l]!='-')//左子树会在下一个循环先出栈
        {
            st[top++] = l;
        }
    }
    free(st);
}
```

# 后序
在前序的方法中，如果先压左子树再压右子树，出来的结果是根右左，再放到栈
里边逆序后成根左右。
```C
void PostOrder(BinaryTree tree)
{
    if (tree == NULL || tree->size < 1)
        return;
    if (tree->data[1] == NIL)
        return;

    int n = tree->size;
    int *s1 = (int *)malloc(sizeof(int) * (n + 1));
    int *s2 = (int *)malloc(sizeof(int) * (n + 1));
    int top1 = -1, top2 = -1;

    s1[++top1] = 1;

    while (top1 != -1) {
        int now = s1[top1--];

        /* 记录“根右左”顺序 */
        s2[++top2] = now;

        int l = 2 * now;
        int r = 2 * now + 1;

        /* 为了先弹出右孩子，先压左，再压右 */
        if (l <= n && tree->data[l] != NIL)
            s1[++top1] = l;
        if (r <= n && tree->data[r] != NIL)
            s1[++top1] = r;
    }

    /* s2 逆序输出，即左右根 */
    while (top2 != -1) {
        int now = s2[top2--];
        putchar(tree->data[now]);
    }

    free(s1);
    free(s2);
}
```
