# bellman-ford  
用于解决单源最短路问题    
特点：可以处理负权边，可以检测从源点可达的负权环，时间复杂度为O(VE)比dijkstra慢，但适用性广    
## 松弛操作  
该算法的核心思想就是松弛操作，即对于边(u,v)，用dist(u)和l(u,v)来尝试更新dist(v)。   
```c++
dist(v)=min(dist(v),dist(u)+l(u,v))
```
感觉本质就是把所有可能经过v的路线全列举出来然后找最小的值。来回循环n遍类似于两层for循环，来找到最正确的答案。每一次循环都会使真正的最短路往前进一层。  
当循环次数达到了n轮以上的时候，可以确认出现了负环（可以判断负环）   
## 板子  
```c++
typedef struct Edge{
    int x,y,v;

} edge[M+1];
int n,m,dist[N+1],pre[N+1];
int shortestpath(int s,int t)
{
    memset(dist,127,sizeof(dist));
    dist[s]=0;
    for(;;)
    {
        bool ok =false;
        for(int i=0;i<=m;i++)
        {
            int x =edge[i].x;
            int y = edge[i].y;
            int v=edge[i].v;
            if(dist[x]<1<<30)
            {
                if(dist[x]+v<dist[y])
                {
                    dist[y]=dist[x]+v;
                    pre[y]=x;
                    ok=true;
                }
            }
            if(!ok){
                break;
            }
        }
    }
    return dist[t];
}
```     
# dijkstra  
## 介绍 
我们要维护一个顶点集合C，满足所有对于集合C的顶点x，我们都已经找到了起点S到x的最短路，此时dist[x]记录的就是最终最短路的长度。   
算法流程如下：  
将C设置为空，将S的距离设置为0，其他顶点距离设置为无穷大  
在每一轮中，将离起点最近的（dist最小，不能无穷大）的还不在C中的顶点加入C，并且用这个点连出去的边通过松弛操作尝试更新其他点的dist。   
当T（如果T存在）或者没有新的点加入C时，算法结束。  

## 板子（优先队列优化时间复杂度O((n+m)logn)）

```c++
//deepseek版

#include <bits/stdc++.h>
using namespace std;

using ll = long long;
const ll INF = (1LL << 62);

// 堆优化 Dijkstra
// g 为邻接表：g[u] = { {v, w}, ... }
// 返回 dist 数组，dist[i] 表示 s 到 i 的最短路
vector<ll> dijkstra(int s, const vector<vector<pair<int, ll>>>& g) {
    int n = (int)g.size();
    vector<ll> dist(n, INF);

    priority_queue<pair<ll, int>, vector<pair<ll, int>>, greater<pair<ll, int>>> pq;

    dist[s] = 0;
    pq.push({0, s});

    while (!pq.empty()) {
        auto [d, u] = pq.top();
        pq.pop();

        // 过期状态，跳过
        if (d != dist[u]) continue;

        for (auto [v, w] : g[u]) {
            if (dist[v] > d + w) {
                dist[v] = d + w;
                pq.push({dist[v], v});
            }
        }
    }

    return dist;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m, s;
    cin >> n >> m >> s;   // n 个点，m 条边，起点 s

    vector<vector<pair<int, ll>>> g(n + 1);

    for (int i = 0; i < m; i++) {
        int u, v;
        ll w;
        cin >> u >> v >> w;

        // 有向边
        g[u].push_back({v, w});

        // 如果是无向边，再加下面这句
        // g[v].push_back({u, w});
    }

    vector<ll> dist = dijkstra(s, g);

    for (int i = 1; i <= n; i++) {
        if (dist[i] == INF) cout << -1 << ' ';
        else cout << dist[i] << ' ';
    }
    cout << '\n';

    return 0;
}
```
注意，无论是什么样的最短路，优先队列中一定要按代价最小的那个从小到大排。优先队列默认先按第一个值从小往大排，然后才是后面的，所以代价放第一个。   
该算法可以用在存在环的图，但不能有负权图    
# floyd 
这个算法可以求图中所有顶点之间的最短路  
使用邻接矩阵存储图：a数组为存储有向图边权的矩阵，若存在从i到j的边，a[i][j]记录这些边的边权最小值，若不存在，则a[i][j]无穷大，对于所有i，a[i][i]=0;
## 代码  
```c++
memset(f,127,sizeof(f));
f[0][i][j]=a[i][j];//for循环把图全写进f中去

for(int k=1;k<=n;k++)
{
    for(int i=1;i<=n;i++)
    {
        for(int j=1;j<=n;j++)
        {
            f[k][i][j]=min(f[k-1][i][j],f[k-1][i][k]+f[k-1][k][j]);
        }
    }
}
//f[k][i][j]表示从i到j，中间可以经过1~k号顶点的最短路径的长度
//从i到j，中间可以经过1~k号节点的路径分为两种：一种是没经过k号节点的，另一种是经过k号节点的。
/*
该算法把从i到j的路程分为两部分，i到k，k到j（这是经过1~k节点的情况，对应f[k-1][i][k]（从i走到k）+f[k-1][k][j]（从k走到j））
f[k-1][i][j]对应着不经过k的情况（动态规划递推？）
f[n][u][v]则为u到v的最短路径长度，若为无穷大，则路不通
时间复杂度O(n^3)
注意f[k-1][i][k]+f[k-1][k][j]可能会爆数据范围


*/
//空间复杂度优化版
memset(v,127,sizeof(v));
v[i][j]=a[i][j];
for(int k = 1;k<=n;k++)
{
    for(int i=1;i<=n;i++)
    {
        for(int j=1;j<=n;j++)
        {
            if(v[i][k]<1<<30&&v[k][j]<1<<30)
            {
                v[i][j]=min(v[i][j],v[i][k]+v[k][j]);
            }

    }

//AI版
#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll INF = 1e18;          // 无穷大，注意不要用 LLONG_MAX，加法会溢出
const int MAXN = 505;         // 节点数上限

int n, m;                     // n 个点，m 条边
ll dist[MAXN][MAXN];          // dist[i][j] 表示 i 到 j 的最短路
int nxt[MAXN][MAXN];          // 路径还原：i 到 j 最短路上的下一个点

void init() {
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            dist[i][j] = (i == j ? 0 : INF);
            nxt[i][j] = (i == j ? i : -1);
        }
    }
}

// 加边（有向边；无向边就再调用一次 addEdge(v, u, w)）
void addEdge(int u, int v, ll w) {
    if (w < dist[u][v]) {
        dist[u][v] = w;
        nxt[u][v] = v;
    }
}

void floyd() {
    for (int k = 1; k <= n; k++) {
        for (int i = 1; i <= n; i++) {
            if (dist[i][k] == INF) continue;        // 剪枝
            for (int j = 1; j <= n; j++) {
                if (dist[k][j] == INF) continue;    // 剪枝
                if (dist[i][k] + dist[k][j] < dist[i][j]) {
                    dist[i][j] = dist[i][k] + dist[k][j];
                    nxt[i][j] = nxt[i][k];
                }
            }
        }
    }
}

// 还原 i -> j 的路径，返回节点序列；不可达返回空
vector<int> getPath(int i, int j) {
    vector<int> path;
    if (nxt[i][j] == -1) return path;      // 不可达
    path.push_back(i);
    while (i != j) {
        i = nxt[i][j];
        path.push_back(i);
    }
    return path;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;
    init();
    for (int i = 0; i < m; i++) {
        int u, v; ll w;
        cin >> u >> v >> w;
        addEdge(u, v, w);                  // 无向图再加 addEdge(v, u, w);
    }

    floyd();

    // 输出示例：任意两点间距离
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            cout << (dist[i][j] == INF ? -1 : dist[i][j]) << " ";
        }
        cout << "\n";
    }
    return 0;
}
```
