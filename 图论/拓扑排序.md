# 使用条件  
有向无环图（拓扑排序也可以检查是否是有向无环图）  
# 性质  
经过拓扑排序后，对于所有边u->v，总有排完序后u都在v前边，且每个点只会出现一次。     
一个图可能有多个拓扑排序结果  
# 用途  
当你的问题可以抽象成“元素间有先后约束，且需要按顺序全部走一遍，同时需要先排除循环冲突”时，就该用拓扑排序了。   
```c++
#include <bits/stdc++.h> // 包含了所有常用库（竞赛写法，工程上建议分开引用）
using namespace std;

class Solution {
public:
    /**
     * 拓扑排序主函数 (Kahn算法)
     * @param n 图中节点的总数 (节点编号默认 0 到 n-1)
     * @param edges 有向边的集合，每条边表示为 [u, v]，含义是 u -> v (u 依赖于 v，或者 u 先于 v)
     * @return 返回一个合法的拓扑序列；如果图中存在环，则返回空数组 {}
     */
    vector<int> topologicalSort(int n, vector<vector<int>>& edges) {
        
        // ------------------------------------------------------------
        // 第一步：数据结构的初始化
        // ------------------------------------------------------------
        // 邻接表 graph: 用于存储每个节点的所有后继节点（即它指向谁）
        // 之所以用 vector<vector<int>>，是因为稀疏图用邻接表比邻接矩阵省空间（O(V+E)）
        vector<vector<int>> graph(n);
        
        // 入度数组 indegree: indegree[i] 表示有多少条边直接指向节点 i
        // 入度为 0 意味着该节点没有任何前置依赖，可以作为“起点”开始执行
        vector<int> indegree(n, 0);
        
        // ------------------------------------------------------------
        // 第二步：建图 并 统计入度 (遍历所有的边)
        // ------------------------------------------------------------
        for (auto &e : edges) { 
            // 这里使用 auto &e 是为了避免拷贝，直接引用原数据，提高效率
            int u = e[0]; 
            int v = e[1];
            
            // 1. 将边存入邻接表：u 执行完后，v 才可以执行
            graph[u].push_back(v);
            
            // 2. 因为 u -> v，所以 v 多了一个“前驱节点”，v 的入度加 1
            indegree[v]++;
        }

        // ------------------------------------------------------------
        // 第三步：初始化队列，将所有“入度为 0”的节点作为起点放入队列
        // ------------------------------------------------------------
        // 这里使用标准队列 queue，保证先进先出（FIFO）。
        // 注意：如果题目要求字典序最小（如按编号从小到大输出），需要改用 priority_queue（见后面进阶版）
        queue<int> q;
        for (int i = 0; i < n; i++) {
            if (indegree[i] == 0) {
                q.push(i); // 这些节点没有依赖，可以最先执行
            }
        }

        // ------------------------------------------------------------
        // 第四步：存储最终结果的数组
        // ------------------------------------------------------------
        vector<int> result;
        // 这里提前 reserve 空间可以避免动态扩容带来的性能损耗，但非必须
        result.reserve(n); 

        // ------------------------------------------------------------
        // 第五步：核心 BFS 循环（层层剥洋葱）
        // ------------------------------------------------------------
        while (!q.empty()) {
            // 1. 取出队首节点 u（表示该节点当前可以被执行了）
            int u = q.front(); 
            q.pop();
            
            // 2. 将 u 放入结果序列（代表安排 u 在当前顺序执行）
            result.push_back(u);
            
            // 3. 遍历 u 的所有后继节点 v（即 u -> v 这条边）
            for (int v : graph[u]) {
                // 关键逻辑：既然 u 已经被执行完了，那么 u 对 v 的“前置约束”就消失了
                // 所以 v 的入度减 1
                indegree[v]--;
                
                // 如果 v 的入度变成了 0，说明 v 的所有前置节点（依赖）都已经执行完毕了
                // 那么 v 现在就没有任何阻碍了，可以入队等待下一轮执行
                if (indegree[v] == 0) {
                    q.push(v);
                }
            }
        }

        // ------------------------------------------------------------
        // 第六步：环的检测（非常重要！）
        // ------------------------------------------------------------
        // 理论依据：如果图中存在环，环上的节点入度永远不可能降到 0（因为互相依赖），
        // 它们永远不会被加入队列，也就不会被 pop 出来放入 result。
        // 因此，如果成功排序的节点数少于总节点数，说明存在环，无法进行拓扑排序。
        if ((int)result.size() != n) {
            return {}; // 返回空数组，表示排序失败（存在循环依赖）
        }
        
        // 排序成功，返回合法的线性序列
        return result;
    }
};
```
```c++
#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    vector<int> topologicalSortLexicographically(int n, vector<vector<int>>& edges) {
        vector<vector<int>> graph(n);
        vector<int> indegree(n, 0);

        // 建图与统计入度（逻辑同上，这里略写）
        for (auto &e : edges) {
            graph[e[0]].push_back(e[1]);
            indegree[e[1]]++;
        }

        // ------------------------------------------------------------
        // 核心改动点：用优先队列（小根堆）替代普通队列
        // 定义：priority_queue<Type, Container, Functional>
        // greater<int> 表示数字小的优先级高，即堆顶永远是当前编号最小的节点
        // ------------------------------------------------------------
        priority_queue<int, vector<int>, greater<int>> pq;
        
        for (int i = 0; i < n; i++) {
            if (indegree[i] == 0) {
                pq.push(i);
            }
        }

        vector<int> result;
        result.reserve(n);

        while (!pq.empty()) {
            // 每次弹出当前可选节点中，编号最小的那个
            int u = pq.top(); 
            pq.pop();
            
            result.push_back(u);

            for (int v : graph[u]) {
                indegree[v]--;
                if (indegree[v] == 0) {
                    pq.push(v); // 这里依然保持小根堆的特性，所有入队的节点会自动排序
                }
            }
        }

        if ((int)result.size() != n) {
            return {}; // 依然要记得检测环
        }
        return result;
    }
};
```

# 拓扑排序+动态规划求有向无环图最长路径  
拓扑排序排好后，按顺序  
dp[v]=max(dp[u]+w[u->v],dp[v])
```C++
//洛谷P1113
#include<bits/stdc++.h>
using namespace std;

int main(void)
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    int n;
    cin>>n;
    vector<vector<int>>mp(n+1,vector<int>(0));
    vector<int>w(n+1,0);
    vector<int>dp(n+1,0);
    vector<int>in(n+1,0);
    for(int i=0;i<n;i++)
    {
        int a,b,c;
        cin>>a>>b;
        w[a]=b;
        cin>>c;
        while(c!=0){
            in[c]++;
            mp[a].push_back(c);
            cin>>c;
        }


    }
    priority_queue<int,vector<int>,greater<int>>pq;
    vector<int>pp;
    for(int i=1;i<=n;i++)
    {
        if(in[i]==0){
            pq.push(i);
        }
    }
    while(!pq.empty()){
        int tt=pq.top();
        pp.push_back(tt);
        pq.pop();
        for(int i=0;i<mp[tt].size();i++)
        {
            in[mp[tt][i]]--;
            if(in[mp[tt][i]]==0){
                pq.push(mp[tt][i]);
            }

        }

    }
    for(int i=0;i<n;i++)
    {
        if(dp[pp[i]]==0){
            dp[pp[i]]=w[pp[i]];
        }
        for(int j=0;j<mp[pp[i]].size();j++){
            dp[mp[pp[i]][j]]=max(dp[pp[i]]+w[mp[pp[i]][j]],dp[mp[pp[i]][j]]);
        }
    }
    int max=0;
    for(int i=1;i<=n;i++){
        if(max<dp[i])
        {
            max=dp[i];
        }
    }
    cout<<max;
}
```
