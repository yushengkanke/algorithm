# 基本概念
一个图是由点集v和边集e组成的,v包含所有点，e包含所有边，v为空为空图   
由无向边构成的图成为无向图，有向边为有向图   
自环：边链接的两个点是同一个点   
重边：无向图中的两点之间有多条边，有向图中两点之间有多条同方向的边   
孤点：没有连接边的点   
简单图：没有自环和重边的图    
度数：对于无向图中的顶点v，v作为边的端点的次数成为v的度数。对于有向图来说，作为起点的次数称之为出度，作为终点的次数称之为入度  
一张图的所有点的度数和为边数的两倍     
完全图：无向图：图中每个顶点都和其余点有边相连，有向图：图中每个顶点都有连向所有其他点的边，也有所有其他点连向自己的边   
# 图的存储

# 网格图的BFS  
```c++
#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> bfs(const vector<string>& grid, int sx, int sy) {
    int n = grid.size();
    int m = grid[0].size();

    // dist[i][j] = -1 表示还没走到
    vector<vector<int>> dist(n, vector<int>(m, -1));

    queue<pair<int, int>> q;
    dist[sx][sy] = 0;   // 起点第 0 步
    q.push({sx, sy});

    int dx[4] = {-1, 1, 0, 0};
    int dy[4] = {0, 0, -1, 1};

    while (!q.empty()) {
        pair<int, int> cur = q.front();
        q.pop();

        int x = cur.first;
        int y = cur.second;

        for (int d = 0; d < 4; d++) {
            int nx = x + dx[d];
            int ny = y + dy[d];

            // 越界
            if (nx < 0 || nx >= n || ny < 0 || ny >= m) continue;

            // 障碍物
            if (grid[nx][ny] == '#') continue;

            // 已经走过
            if (dist[nx][ny] != -1) continue;

            // 标记最早步数，并入队
            dist[nx][ny] = dist[x][y] + 1;
            q.push({nx, ny});
        }
    }

    return dist;
}

int main() {
    vector<string> grid = {
        "....",
        ".#..",
        "...."
    };

    int sx = 0, sy = 0; // 起点

    vector<vector<int>> dist = bfs(grid, sx, sy);

    for (int i = 0; i < (int)dist.size(); i++) {
        for (int j = 0; j < (int)dist[i].size(); j++) {
            cout << setw(3) << dist[i][j];
        }
        cout << '\n';
    }

    return 0;
}
```
就是在标记点在哪一步被走到时，在入队的时候标记，不要再出队的时候想办法找就行了。
