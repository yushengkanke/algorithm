# 中国剩余定理(互质) 
![](/images/数论/11.png)  
条件：  
当模数们互质的时候  
结论：  
x有解，并且在m1\*m2\*m3...*mn的范围中有唯一解  
x的求解公式如下图  
![](/images/数论/12.png)  
这只是x的一个特殊解，事实上，把x+或-模数们的最小公倍数都是解  
```c++
// 扩展欧几里得
int exgcd(int a, int b, int &x, int &y) {
    if (b == 0) { x = 1; y = 0; return a; }
    int d = exgcd(b, a % b, y, x);
    y -= (a / b) * x;
    return d;
}

// 计算逆元
int modInverse(int a, int m) {
    int x, y;
    exgcd(a, m, x, y);
    x %= m;
    if (x < 0) x += m;
    return x;
}

// 中国剩余定理主函数
// 输入：模数数组 m，余数数组 a
// 返回 x，满足 x ≡ a[i] (mod m[i])
long long CRT(const vector<int>& m, const vector<int>& a) {
    long long M = 1;
    int n = m.size();
    for (int i = 0; i < n; ++i) M *= m[i];

    long long x = 0;
    for (int i = 0; i < n; ++i) {
        long long Mi = M / m[i];
        long long yi = modInverse(Mi % m[i], m[i]);
        x = (x + Mi * yi % M * a[i] % M) % M;
    }
    return (x + M) % M;
}
```

```c++
using i64 = long long;
i64 exgcd(i64 a, i64 b, i64 &x, i64 &y) {
    if (!b) { x = 1; y = 0; return a; }
    i64 d = exgcd(b, a % b, y, x);
    y -= a / b * x;
    return d;
}
i64 mul(i64 a, i64 b, i64 mod) { // 龟速乘，防止两个1e18相乘溢出
    i64 res = 0;
    while (b) {
        if (b & 1) res = (res + a) % mod;
        a = (a + a) % mod;
        b >>= 1;
    }
    return res;
}

i64 EXCRT(vector<i64> a, vector<i64> m) {
    i64 A = a[0], M = m[0]; // 初始方程
    for (int i = 1; i < a.size(); i++) {
        i64 g = exgcd(M, m[i], x, y);
        i64 c = ((a[i] - A) % m[i] + m[i]) % m[i];
        
        if (c % g != 0) return -1; // 无解
        
        i64 mod = m[i] / g;
        // 解出 t = x * (c/g) mod (mod)
        // 注意这里要用龟速乘，x可能很大
        // 可以直接用乘法，改__int128(数据范围M小于1e18的话直接用long long),龟速乘会慢,它是以前没有128位时的产物
        i64 t = mul(x, c / g, mod); 
        
        // 更新 A 和 M
        A = A + mul(t, M, M * mod); // 注意这里模数是新的M，但M*mod可能爆long long？谨慎处理
        // 更安全的写法：先用M存旧值，再更新
        M = M * mod; 
        A = (A % M + M) % M;
    }
    return A;
}
```
