# 线性筛  
直接AI（省事这一块）  
埃氏筛：其实就是往后枚举质数的倍数，做好标记，遍历时没被标记的就是质数。  
```c++
#include <bits/stdc++.h>
using namespace std;

const int N = 1e6 + 5;

int primes[N], cnt;          // 存储素数
bool st[N];                  // st[i] = true 表示 i 被筛掉（非素数）
int phi[N];                  // 欧拉函数
int mu[N];                   // 莫比乌斯函数

// 线性筛素数 + 计算欧拉函数 + 莫比乌斯函数
void linear_sieve(int n) {
    phi[1] = 1;
    mu[1] = 1;
    for (int i = 2; i <= n; ++i) {
        if (!st[i]) {                // i 是素数
            primes[cnt++] = i;
            phi[i] = i - 1;
            mu[i] = -1;
        }
        for (int j = 0; j < cnt && i * primes[j] <= n; ++j) {
            int x = i * primes[j];
            st[x] = true;
            if (i % primes[j] == 0) { // primes[j] 是 i 的最小质因子
                phi[x] = phi[i] * primes[j];
                mu[x] = 0;
                break;
            } else {                  // 互质
                phi[x] = phi[i] * (primes[j] - 1);
                mu[x] = -mu[i];
            }
        }
    }
}

// 如果只需要素数，去掉 phi 和 mu 相关代码
// 如果只需要 phi，只保留 phi 相关
// 如果只需要 mu，只保留 mu 相关
```
# 整除分块

```c++
long long divide_block(long long n) {
    long long ans = 0;
    for (long long l = 1, r; l <= n; l = r + 1) {
        long long v = n / l;      // 当前区间的值
        r = n / v;                // 区间右端点
        ans += (r - l + 1) * v;
    }
    return ans;
}
```
