```c++
int gcd (int a,int b)
{
    if(b==0)return a;
    else return gcd(b,a%b);
}
```  

# 扩展欧几里得  
对于求解a*x+b*y=1的问题(a大b小,gcd(a,b)=1)，我们可以  
先把算式拆开变成b*(a/b*x+y)+a%b=1  
然后换元成b*x1+a*y1=1  
最后递归到y的系数为一时令y为1求解  
![](/images/数论/4.png)  
```c++
int exgcd(int a , int b , int& x , int& y){
    if(b==0){
        x=1;
        y=0;
        return a;
    }
    int xx,yy;
    int d=exgcd(b,a%b,xx,yy);
    x=yy;
    y=xx-(a/b)*yy;
    return d;
}
//其中return的d代表(原gcd(a,b))。x,y可以先声明然后通过这个自动有正确的值
//结果的x<a,y<b
//若等式右边不是1而是p,则最终答案x=x*p,y=y*p
//若要求x,y的所有解，则用以下思路
//在a*x+b*y=p的前提下，x减少b*t个，则y应增加a*t个，才能维持等式。
//若要解出x的最小非负整数解，则应在结果算出后加入(因为出来的解有可能是复数)
b=-b;
while(a<0||b<0)x+=b/d,y+=a/d;//变大
while(x>=b/d&&y>=a/d)x-=b/d,y-=a/d;//变小

```
# 贝祖等式(exgcd)  
exgcd的本质  
在上文的a,b为一般数时，等式为a*x+b*y=gcd(a,b)  
这玩意能在正常求gcd的同时还能提供一个解，所以叫exgcd  
```C++
i64 exgcd(i64 a, i64 b, i64 &x, i64 &y) {
    if (!b) {          // 递归终点：b == 0
        x = 1; 
        y = 0; 
        return a;      // 此时 a 就是 gcd(a, 0)，直接返回 a
    }
    i64 d = exgcd(b, a % b, y, x); // 递归求 gcd(b, a%b)
    y -= a / b * x;
    return d;          // 原样返回每一层递归传上来的 gcd
}


using i64 = long long;

// 返回值 d 是 gcd(a, b)
// 引用传参 x, y，函数结束后满足 a*x + b*y = d
i64 exgcd(i64 a, i64 b, i64 &x, i64 &y) {
    if (!b) {          // 递归终点：此时 a = gcd, b = 0
        x = 1; 
        y = 0;         // 因为 a*1 + 0*0 = a = gcd
        return a;
    }
    i64 x1, y1;
    i64 d = exgcd(b, a % b, x1, y1); // 向下递归
    
    // 回溯更新（套用上面的数学推导公式）
    x = y1;
    y = x1 - (a / b) * y1;
    return d;
}
```
