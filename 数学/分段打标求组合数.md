# 分段打标法  
适用于像求组合数这样的问题，a[n]与a[n-i]项有关系的问题，且n非常大。  
首先应该打表  
```c++
#include<bits/stdc++.h>
using namespace std;

typedef long long ll;
const int mod =1e9+7;
int main(void)
{
    ll fac =1;
    for(int i=1;i<mod ;i++)
    {
        fac=fac*i%mod;
        if(i%100000==0)
        printf("%lld,",fac);
        //这里的1000000是你打表的间隔，输出中要加，方便复制粘贴进数组
    }
}
```
然后在正式提交中使用  
```c++
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int T=1000000//间隔
const int mod=1e9+7;
ll fact[]={
    //把上边的答案复制进来
} 
ll qpow(ll a, ll b) {
    ll res = 1;
    while (b) {
        if (b & 1) res = res * a % MOD;
        a = a * a % MOD;
        b >>= 1;
    }
    return res;
}
int fac (int n)//求阶乘
{
    ll v =fact[n/T]；
    for(int i =n/T*T+1;i<=n;i++)
    {
        v=v*i%mod;
    }
    return v;
}
ll C(int n, int k) {
    if (k < 0 || k > n) return 0;
    return fac(n)*qpow(fac(k)*fac(n-k)%mod,mod-2)%mod;
}
int main(void)
{
    int t;
    scanf("%d",&t);
    for(int tc=0;tc<t;tc++)
    {
        int a,b;
        scanf("%d%d",&a,&b);
        printf("%lld",C(a,b));
    }
}
```
其他的类型如果要用分段打表法做法类似。就是本来应该从1到n开始算的东西，预先平均分段，  然后再从最近的分好的点开始递推。
